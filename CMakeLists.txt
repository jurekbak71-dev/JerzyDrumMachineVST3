cmake_minimum_required(VERSION 3.22)
project(JerzyDrumMachine VERSION 0.1.0)
set(CMAKE_CXX_STANDARD 17)
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.4)
FetchContent_MakeAvailable(JUCE)
juce_add_plugin(JerzyDrumMachine COMPANY_NAME "Jerzy Audio" BUNDLE_ID "com.jerzyaudio.drummachine" PLUGIN_MANUFACTURER_CODE Jrzy PLUGIN_CODE JDrm FORMATS VST3 Standalone PRODUCT_NAME "Jerzy Drum Machine" IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
target_sources(JerzyDrumMachine PRIVATE Source/PluginProcessor.cpp Source/PluginProcessor.h Source/PluginEditor.cpp Source/PluginEditor.h Source/DrumEngine.h)
target_compile_definitions(JerzyDrumMachine PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(JerzyDrumMachine PRIVATE juce::juce_audio_utils juce::juce_dsp PUBLIC juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
